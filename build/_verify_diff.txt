+/* ---------------- ★激光"连续确认"（2026-09-26，立柱前校准用）----------------
+   LASER_Barrier() 内部已有"5ms 内连续 3 次低电平"的引脚级消抖，但车在慢速移动、柱子边缘反光时，
+   偶尔仍会冒出一两拍假的"有障碍物"。下面这个是在它之上再加一层**时间维度**的确认：
+   连续 count 次调用 LASER_Barrier() 都为 1(有障碍物)才返回 1；中途任何一次读到 0 就立刻返回 0(重新数)。
+   ★用法：当 WAIT_WHILE 的条件用 —— 返回 0 就继续等(车继续慢移)、返回 1 才算真到位；
+     这样 6s 超时兜底和条件循环里的串口数据泵都保留。
+   ★代价：确认一次最多 (count-1)×gap ms（默认 3 次→约 40ms），5cm/s 下多走约 2mm，可忽略。 */
+#define LASER_CONFIRM_GAP_MS      20U   //连续确认时两次采样之间的间隔(ms)
+#define LIZHU_LASER4_CONFIRM_CNT  3U    //★立柱前"左激光(LASER4)看到柱子"要连续确认几次才算数
+static uint8_t LASER_BarrierConfirm(GPIO_TypeDef *GPIOx, uint16_t GPIO_Pin, uint8_t count){
+  if(count == 0U) return LASER_Barrier(GPIOx, GPIO_Pin);      //防御：次数给 0 时退化成单次读取
+  for(uint8_t i = 0; i < count; i++){
+    if(LASER_Barrier(GPIOx, GPIO_Pin) == 0U) return 0U;       //这一拍没障碍物 → 不算数，从头再数
+    if((uint8_t)(i + 1U) < count) HAL_Delay(LASER_CONFIRM_GAP_MS);
+  }
+  return 1U;                                                  //连续 count 次都有障碍物 → 确认
+}
